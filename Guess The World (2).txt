        function createSeededRandom(seedStr) {
            let h = 2166136261 >>> 0;
            for (let i = 0; i < seedStr.length; i++) {
                h = Math.imul(h ^ seedStr.charCodeAt(i), 16777619);
            }
            return function() {
                h += 0x6D2B79F5;
                let t = Math.imul(h ^ (h >>> 15), 1 | h);
                t ^= t + Math.imul(t ^ (t >>> 7), 61 | t) ^ t;
                return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
            };
        }

        function initDailyChallenge() {
            const now = new Date();
            const dateStr = now.toISOString().split('T')[0];
            document.getElementById('daily-date').innerText = `CHALLENGE DU ${dateStr}`;

            const rng = createSeededRandom(dateStr);
            const pool = [...LOCATION_DATABASE];
            dailyCountries = [];

            for (let i = 0; i < 5; i++) {
                if (pool.length === 0) break;
                const idx = Math.floor(rng() * pool.length);
                dailyCountries.push(pool.splice(idx, 1)[0]);
            }

            // Load saved state from LocalStorage
            const savedState = localStorage.getItem(`guess_world_${dateStr}`);
            if (savedState) {
                const parsed = JSON.parse(savedState);
                roundResults = parsed.roundResults || [];
                currentRoundIndex = roundResults.length;
                totalDailyScore = parsed.totalDailyScore || 0;
                streak = parsed.streak || 0;
            }

            updateStreakDisplay();
            updateStatsUI();

            if (currentRoundIndex >= 5) {
                renderEndGameGlobeRecap();
                showSummaryModal();
            } else {
                setupRound(currentRoundIndex);
            }
        }

        function calculateHaversine(lat1, lon1, lat2, lon2) {
            const R = 6371; // Earth radius in km
            const dLat = (lat2 - lat1) * Math.PI / 180;
            const dLon = (lon2 - lon1) * Math.PI / 180;
            const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
                      Math.sin(dLon/2) * Math.sin(dLon/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
            return Math.round(R * c);
        }

        function getCardinalDirection(lat1, lon1, lat2, lon2) {
            const dLon = (lon2 - lon1) * Math.PI / 180;
            const y = Math.sin(dLon) * Math.cos(lat2 * Math.PI / 180);
            const x = Math.cos(lat1 * Math.PI / 180) * Math.sin(lat2 * Math.PI / 180) -
                      Math.sin(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.cos(dLon);
            let brng = Math.atan2(y, x) * 180 / Math.PI;
            brng = (brng + 360) % 360;

            const dirsFR = ["Nord ⬆️", "Nord-Est ↗️", "Est ➡️", "Sud-Est ↘️", "Sud ⬇️", "Sud-Ouest ↙️", "Ouest ⬅️", "Nord-Ouest ↖️"];
            const dirsEN = ["North ⬆️", "North-East ↗️", "East ➡️", "South-East ↘️", "South ⬇️", "South-West ↙️", "West ⬅️", "North-West ↖️"];
            const idx = Math.round(brng / 45) % 8;
            return currentLang === 'fr' ? dirsFR[idx] : dirsEN[idx];
        }

        function confirmGuess() {
            if (!userGuessCoords) return;

            const target = dailyCountries[currentRoundIndex];
            const dist = calculateHaversine(userGuessCoords.lat, userGuessCoords.lng, target.coords.lat, target.coords.lng);
            const dir = getCardinalDirection(userGuessCoords.lat, userGuessCoords.lng, target.coords.lat, target.coords.lng);

            // Maximum points based on clue level used
            const maxPts = 1000 - (currentClueLevel - 1) * 200;
            let score = 0;
            let isInsideCountry = false;

            if (target.type === 'country') {
                // RULE: If guess lands INSIDE country territory radius, award MAXIMUM score for clue level!
                if (dist <= target.radiusKm) {
                    isInsideCountry = true;
                    score = maxPts;
                } else {
                    const extraDist = dist - target.radiusKm;
                    score = Math.floor(maxPts * Math.exp(-extraDist / 1200));
                }
            } else {
                // City target radius
                if (dist <= 25) {
                    isInsideCountry = true;
                    score = maxPts;
                } else {
                    score = Math.floor(maxPts * Math.exp(-dist / 1200));
                }
            }

            // Draw target pin and connecting arc curve
            setTargetMarker(target.coords.lat, target.coords.lng);
            drawArcBetweenPoints(userGuessCoords, target.coords);
            focusGlobeOn(target.coords.lat, target.coords.lng);

            // Record round
            const roundData = {
                target,
                guessCoords: userGuessCoords,
                distance: dist,
                score,
                isInsideCountry,
                cluesUsed: currentClueLevel
            };
            roundResults.push(roundData);
            totalDailyScore += score;

            // Show Result Modal
            document.getElementById('result-flag').innerText = target.flag;
            document.getElementById('result-name').innerText = target.names[currentLang];
            document.getElementById('result-type').innerText = target.type === 'country' ? 'PAYS CIBLE' : 'VILLE CIBLE';
            document.getElementById('result-dist').innerText = `${dist.toLocaleString()} km`;
            document.getElementById('result-dir').innerText = dir;
            document.getElementById('result-score').innerText = `+${score} pts`;
            document.getElementById('result-capital').innerText = target.capital;
            document.getElementById('result-pop').innerText = target.pop;
            document.getElementById('result-fact').innerText = `"${target.fact[currentLang]}"`;

            const insideBadge = document.getElementById('result-inside-badge');
            if (isInsideCountry) {
                insideBadge.classList.remove('hidden');
            } else {
                insideBadge.classList.add('hidden');
            }

            openModal('result-modal');

            // Save Progress
            const now = new Date();
            const dateStr = now.toISOString().split('T')[0];
            if (roundResults.length === 5) {
                streak++;
            }
            localStorage.setItem(`guess_world_${dateStr}`, JSON.stringify({
                roundResults,
                totalDailyScore,
                streak
            }));
            updateStreakDisplay();
        }

        let scene, camera, renderer, controls;
        let globeMesh;
        let recapMarkersGroup = new THREE.Group();
        let bordersGroup = new THREE.Group();
        let raycaster, mouse;
        const GLOBE_RADIUS = 5;
        let worldGeoData = null;

        async function loadCountryBorders() {
            try {
                const res = await fetch('https://cdn.jsdelivr.net/npm/world-atlas@2/countries-110m.json');
                worldGeoData = await res.json();
            } catch (err) {
                console.warn("Could not load country borders topology", err);
            }
        }

        function drawWorldCountryBorders() {
            while (bordersGroup.children.length > 0) {
                bordersGroup.remove(bordersGroup.children[0]);
            }
            if (!worldGeoData) return;

            // Draw full country boundary mesh line geometries
            const mesh = topojson.mesh(worldGeoData, worldGeoData.objects.countries);
            if (mesh && mesh.coordinates) {
                const material = new THREE.LineBasicMaterial({
                    color: 0x38bdf8,
                    transparent: true,
                    opacity: 0.55
                });

                mesh.coordinates.forEach(lineCoords => {
                    const points = [];
                    lineCoords.forEach(([lng, lat]) => {
                        points.push(latLngToVector3(lat, lng, GLOBE_RADIUS, 0.018));
                    });
                    if (points.length > 1) {
                        const geom = new THREE.BufferGeometry().setFromPoints(points);
                        bordersGroup.add(new THREE.Line(geom, material));
                    }
                });
            }
        }

        function init3DGlobe() {
            const container = document.getElementById('globe-container');

            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x05070e);
            scene.add(recapMarkersGroup);
            scene.add(bordersGroup);

            camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.set(0, 0, 15);

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true, powerPreference: "high-performance" });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.minDistance = 6.2;
            controls.maxDistance = 25;
            controls.autoRotate = true;
            controls.autoRotateSpeed = 0.4;

            // Lighting
            const ambient = new THREE.AmbientLight(0xffffff, 0.6);
            scene.add(ambient);

            const sun = new THREE.DirectionalLight(0xffffff, 1.2);
            sun.position.set(12, 10, 10);
            scene.add(sun);

            // Clean Globe Texture
            const geometry = new THREE.SphereGeometry(GLOBE_RADIUS, 128, 128);
            const textureLoader = new THREE.TextureLoader();
            const earthTexture = textureLoader.load('https://unpkg.com/three-globe/example/img/earth-blue-marble.jpg');

            const globeMaterial = new THREE.MeshStandardMaterial({
                map: earthTexture,
                roughness: 0.7,
                metalness: 0.1
            });

            globeMesh = new THREE.Mesh(geometry, globeMaterial);
            scene.add(globeMesh);

            // Atmosphere Glow
            const atmosGeo = new THREE.SphereGeometry(GLOBE_RADIUS * 1.12, 64, 64);
            const atmosMat = new THREE.ShaderMaterial({
                vertexShader: `
                    varying vec3 vNormal;
                    void main() {
                        vNormal = normalize(normalMatrix * normal);
                        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
                    }
                `,
                fragmentShader: `
                    varying vec3 vNormal;
                    void main() {
                        float intensity = pow(0.6 - dot(vNormal, vec3(0, 0, 1.0)), 2.0);
                        gl_FragColor = vec4(0.2, 0.6, 1.0, 1.0) * intensity;
                    }
                `,
                blending: THREE.AdditiveBlending,
                side: THREE.BackSide,
                transparent: true
            });
            scene.add(new THREE.Mesh(atmosGeo, atmosMat));

            raycaster = new THREE.Raycaster();
            mouse = new THREE.Vector2();

            window.addEventListener('resize', onWindowResize);
            window.addEventListener('pointerdown', onGlobePointerDown);

            animate();
        }

        function latLngToVector3(lat, lng, radius, height = 0) {
            const phi = (90 - lat) * (Math.PI / 180);
            const theta = (lng + 180) * (Math.PI / 180);
            const r = radius + height;
            return new THREE.Vector3(
                -(r * Math.sin(phi) * Math.cos(theta)),
                r * Math.cos(phi),
                r * Math.sin(phi) * Math.sin(theta)
            );
        }

        function vector3ToLatLng(vec) {
            const radius = vec.length();
            const lat = 90 - (Math.acos(vec.y / radius) * 180 / Math.PI);
            const lng = ((Math.atan2(vec.z, -vec.x) * 180 / Math.PI) - 180);
            return { lat, lng };
        }

        let pointerDownPos = { x: 0, y: 0 };
        function onGlobePointerDown(event) {
            pointerDownPos = { x: event.clientX, y: event.clientY };
            window.addEventListener('pointerup', onGlobePointerUp, { once: true });
        }

        function onGlobePointerUp(event) {
            if (currentRoundIndex >= 5) return; // Game over recap mode

            const distMoved = Math.hypot(event.clientX - pointerDownPos.x, event.clientY - pointerDownPos.y);
            if (distMoved > 6) return;

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObject(globeMesh);

            if (intersects.length > 0) {
                const hitPoint = intersects[0].point;
                userGuessCoords = vector3ToLatLng(hitPoint);
                
                // Clear active round single guess marker
                clearGlobeMarkers();
                setGuessMarker(userGuessCoords.lat, userGuessCoords.lng, 0x38bdf8);

                document.getElementById('guess-status').innerText = currentLang === 'fr' ?
                    `📍 Marqueur placé ! Cliquez sur Valider.` : `📍 Pin placed! Tap Confirm.`;
                document.getElementById('confirm-guess-btn').disabled = false;
            }
        }

        function setGuessMarker(lat, lng, colorHex = 0x38bdf8, labelText = "") {
            const pos = latLngToVector3(lat, lng, GLOBE_RADIUS, 0.08);
            const pinGroup = new THREE.Group();

            const sphere = new THREE.Mesh(
                new THREE.SphereGeometry(0.12, 16, 16),
                new THREE.MeshBasicMaterial({ color: colorHex })
            );
            sphere.position.copy(pos);

            const lineGeo = new THREE.BufferGeometry().setFromPoints([
                latLngToVector3(lat, lng, GLOBE_RADIUS),
                latLngToVector3(lat, lng, GLOBE_RADIUS, 0.28)
            ]);
            const line = new THREE.Line(lineGeo, new THREE.LineBasicMaterial({ color: colorHex }));

            pinGroup.add(sphere);
            pinGroup.add(line);
            recapMarkersGroup.add(pinGroup);
        }

        function setTargetMarker(lat, lng) {
            const pos = latLngToVector3(lat, lng, GLOBE_RADIUS, 0.08);
            const pinGroup = new THREE.Group();

            const sphere = new THREE.Mesh(
                new THREE.SphereGeometry(0.12, 16, 16),
                new THREE.MeshBasicMaterial({ color: 0x22c55e })
            );
            sphere.position.copy(pos);

            const lineGeo = new THREE.BufferGeometry().setFromPoints([
                latLngToVector3(lat, lng, GLOBE_RADIUS),
                latLngToVector3(lat, lng, GLOBE_RADIUS, 0.3)
            ]);
            const line = new THREE.Line(lineGeo, new THREE.LineBasicMaterial({ color: 0x22c55e }));

            pinGroup.add(sphere);
            pinGroup.add(line);
            recapMarkersGroup.add(pinGroup);
        }

        function drawArcBetweenPoints(c1, c2, colorHex = 0xf59e0b) {
            const v1 = latLngToVector3(c1.lat, c1.lng, GLOBE_RADIUS, 0.05);
            const v2 = latLngToVector3(c2.lat, c2.lng, GLOBE_RADIUS, 0.05);

            const points = [];
            const count = 40;
            for (let i = 0; i <= count; i++) {
                const t = i / count;
                const vec = new THREE.Vector3().lerpVectors(v1, v2, t);
                const altitude = Math.sin(t * Math.PI) * 0.8;
                vec.normalize().multiplyScalar(GLOBE_RADIUS + 0.05 + altitude);
                points.push(vec);
            }

            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({ color: colorHex, linewidth: 2 });
            const arc = new THREE.Line(geometry, material);
            recapMarkersGroup.add(arc);
        }

        function clearGlobeMarkers() {
            while (recapMarkersGroup.children.length > 0) {
                const obj = recapMarkersGroup.children[0];
                recapMarkersGroup.remove(obj);
            }
        }

        function focusGlobeOn(lat, lng) {
            controls.autoRotate = false;
            const targetPos = latLngToVector3(lat, lng, 12);
            camera.position.copy(targetPos);
            camera.lookAt(0, 0, 0);
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }

        window.onload = () => {
            init3DGlobe();
            loadCountryBorders().then(() => {
                drawWorldCountryBorders();
            });
            checkUserAuth();
        };
